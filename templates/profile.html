API 调用示例 
# 提交下载（types 可选 video、audio）
curl -X POST {{ server_url }}/api/add_task \
  -H "Authorization: Bearer $DROPLOAD_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"url": "https://www.youtube.com/watch?v=...", "types": ["video"]}'

# 查询任务状态
curl -X POST {{ server_url }}/api/task_info \
  -H "Authorization: Bearer $DROPLOAD_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"tasks": ["v20260917..."]}'

# 读取下载日志（{% if current_user.role == 'admin' %}管理员可读完整日志{% else %}只包含你的任务{% endif %}；用返回的 cursor 与 file_id 增量读取）
curl {{ server_url }}/api/downloader_log \
  -H "Authorization: Bearer $DROPLOAD_TOKEN"
